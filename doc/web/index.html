快速开始

几行代码即可运行你的第一个流水线

main.go 
package main

import (
    "context"
    "fmt"
    "github.com/LerkoX/flowx"
)

func main() {
    ctx := context.Background()
    rt := flowx.NewRuntime(ctx)

    config := `
Version: "1.0"
Name: hello-pipeline

Executors:
  local:
    type: local
    config:
      shell: bash

Graph: |
  stateDiagram-v2
    [*] --> Hello
    Hello --> [*]

Nodes:
  Hello:
    executor: local
    steps:
      - name: greet
        run: echo "Hello, FlowX!"
`

    p, err := rt.RunSync(ctx, "hello-001", config, nil)
    if err != nil {
        panic(err)
    }
    fmt.Println("Status:", p.Status())
}
