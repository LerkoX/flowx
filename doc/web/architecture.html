flowx/
├── core/                 # 核心类型定义
│   ├── config.go         # 配置结构体定义

│   ├── field.go          # FieldItem 定义
│   └── uuid.go           # UUID 工具函数
├── dag/                  # DAG 流水线核心
│   ├── pipeline.go       # Pipeline 接口定义
│   ├── pipeline_impl.go  # Pipeline + DAG 图实现
│   ├── node.go           # Node 接口定义
│   ├── node_impl.go      # Node 实现
│   ├── edge.go           # Edge 接口定义
│   ├── edge_impl.go      # Edge 实现（条件边）
│   ├── eval_context.go   # 表达式求值上下文
│   ├── extractor.go      # 输出提取器
│   └── snapshot.go       # 快照与恢复
├── executor/             # 执行器系统
│   ├── interfaces.go     # Executor/Adapter/Bridge 接口
│   ├── provider/
│   │   └── provider.go   # Executor 提供者（工厂模式）
│   ├── docker/           # Docker 执行器
│   ├── local/            # Local 执行器
│   └── kubernetes/       # Kubernetes 执行器
├── template/             # 模板引擎
│   ├── template.go       # 模板引擎接口
│   └── template_impl.go  # pongo2 模板引擎实现
├── metadata/             # 元数据存储
│   ├── metadata.go       # MetadataStore 接口
│   └── metadata_impl.go  # 三种存储实现
├── logger/               # 日志系统
│   ├── logger.go         # 日志接口与类型
│   └── console_pusher.go # 控制台日志推送
├── doc/                  # 文档目录
│   └── web/              # HTML 文档站点
├── test/                 # 测试
└── config.example.yaml   # 示例配置文件

核心接口

Runtime 接口

type Runtime interface {
    Get(id string) (Pipeline, error)
    Cancel(ctx context.Context, id string) error
    RunAsync(ctx, id, configYAML string, listener Listener) (Pipeline, error)
    RunSync(ctx, id, configYAML string, listener Listener) (Pipeline, error)
    Rm(id string)
    Done() chan struct{}
    Notify(data any) error
    Ctx() context.Context
    StopBackground()
    StartBackground()
    SetPusher(pusher logger.Pusher)
    SetTemplateEngine(engine TemplateEngine)
    GetTemplateEngine() TemplateEngine
    ExportConfig(id string) (string, error)
    Pause(ctx context.Context, id string) error
    Resume(ctx context.Context, id string) error
    UpdateConfig(ctx context.Context, id string, newConfigYAML string) error
    ModifyGraph(ctx context.Context, id string, modifications GraphModifications) error
}

Pipeline 接口

type Pipeline interface {
    Id() string
    GetGraph() Graph
    SetGraph(graph Graph)
    Status() string
    SetMetadata(store MetadataStore)
    Metadata() Metadata
    Listening(listener Listener)
    Done() <-chan struct{}
    Run(ctx context.Context) error
    Notify()
    Cancel()
    SetExecutorProvider(provider ExecutorProvider)
    SetTemplateEngine(engine TemplateEngine)
    GetTemplateEngine() TemplateEngine
    SetPusher(pusher logger.Pusher)
    Pause() error
    Resume(ctx context.Context) error
    IsModifiable() bool
}

Executor 接口

type Executor interface {
    Prepare(ctx context.Context) error
    Destruction(ctx context.Context) error
    Transfer(ctx context.Context, resultChan chan<- any, commandChan <-chan any, inputChan <-chan []byte)
}

    subgraph RuntimePhase["1. Runtime.RunSync/RunAsync"]
        R1[解析 YAML 配置] --> R2[渲染 Param]
        R2 --> R3[渲染 Metadata.data]
        R3 --> R4[构建 DAG 图]
        R4 --> R5[初始化元数据存储]
        R5 --> R6[创建 Pipeline 实例]
        R6 --> R7[执行 Pipeline.Run]
    end

    subgraph PipelineRun["2. Pipeline.Run"]
        P1[触发 PipelineStart 事件] --> P2[逐层 BFS 执行]
        P2 --> P3[并发执行当前层级节点]
        P3 --> P4[评估出边条件]
        P4 --> P5{回边条件?}
        P5 -->|true| P2
        P5 -->|false| P6[触发 PipelineFinish 事件]
        P6 --> P7[关闭 doneChan]
    end

    subgraph NodeExec["3. 节点执行"]
        N1[模板引擎渲染 run 命令] --> N2[Executor.Transfer]
        N2 --> N3[实时流式输出到日志]
        N3 --> N4[输出提取器提取数据]
        N4 --> N5[(MetadataStore)]
    end

    R7 --> P1
    P3 --> N1
